cmake_minimum_required(VERSION 3.22)
project(voicebox_ming LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the build directory out of DWARF and the ELF build ID of release candidates.
add_compile_options(-fdebug-compilation-dir=.)
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT_LIBRARY}")
foreach(kind IN ITEMS synthesize reference)
    if(kind STREQUAL "synthesize")
        set(source "${MING_NATIVE}/main.cpp")
    else()
        set(source "${MING_SOURCE}/native/reference_encode.cpp")
    endif()
    add_library(ming-${kind} SHARED "${source}" bridge.cpp)
    target_include_directories(ming-${kind} PRIVATE "${ORT_INCLUDE}" "${POCKETFFT_INCLUDE}"
        "${MING_SOURCE}/native" "${MING_SOURCE}/native/third_party")
    target_compile_definitions(ming-${kind} PRIVATE main=ming_cli_main BRIDGE_METHOD=Java_dev_ming_local_Native_${kind})
    target_compile_options(ming-${kind} PRIVATE -O2 -fno-fast-math -ffp-contract=off -fvisibility=hidden)
endforeach()
add_library(ming-ort SHARED ort_bridge.cpp)
target_include_directories(ming-ort PRIVATE "${ORT_INCLUDE}")
target_link_libraries(ming-ort PRIVATE dl)
target_compile_definitions(ming-synthesize PRIVATE OrtGetApiBase=VoiceBoxMingOrtGetApiBase)
target_link_libraries(ming-synthesize PRIVATE ming-ort dl)
target_link_libraries(ming-reference PRIVATE onnxruntime)

add_library(ming-gpu-probe SHARED bridge.cpp)
target_include_directories(ming-gpu-probe PRIVATE "${ORT_INCLUDE}")
target_compile_definitions(ming-gpu-probe PRIVATE VOICEBOX_GPU_PROBE)
target_link_libraries(ming-gpu-probe PRIVATE onnxruntime log)

add_library(ming-htp SHARED htp_bridge.cpp bridge.cpp)
target_include_directories(ming-htp PRIVATE "${ORT_INCLUDE}" "${MING_NATIVE}" "${MING_SOURCE}/native")
target_compile_definitions(ming-htp PRIVATE ORT_API_MANUAL_INIT OrtGetApiBase=VoiceBoxMingOrtGetApiBase)
target_compile_options(ming-htp PRIVATE -fvisibility=hidden)
target_link_libraries(ming-htp PRIVATE ming-ort log)

foreach(kind IN ITEMS neon bf16)
    if(kind STREQUAL "neon")
        set(source portable_mmla_ops.cpp)
        set(arch armv8-a)
    else()
        set(source mmla_ops.cpp)
        set(arch armv8-a+bf16)
    endif()
    add_library(ming-mmla-${kind} SHARED "${MING_SOURCE}/native/${source}")
    target_include_directories(ming-mmla-${kind} PRIVATE "${ORT_INCLUDE}")
    target_compile_options(ming-mmla-${kind} PRIVATE -O3 -march=${arch} -fno-fast-math -ffp-contract=off)
endforeach()
