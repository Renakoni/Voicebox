cmake_minimum_required(VERSION 3.22)
project(voicebox_ming LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT_LIBRARY}")
foreach(kind IN ITEMS synthesize reference)
    if(kind STREQUAL "synthesize")
        set(source main.cpp)
    else()
        set(source reference_encode.cpp)
    endif()
    add_library(ming-${kind} SHARED "${MING_SOURCE}/native/${source}" bridge.cpp)
    target_include_directories(ming-${kind} PRIVATE "${ORT_INCLUDE}" "${POCKETFFT_INCLUDE}" "${MING_SOURCE}/native/third_party")
    target_compile_definitions(ming-${kind} PRIVATE main=ming_cli_main BRIDGE_METHOD=Java_dev_ming_local_Native_${kind})
    target_compile_options(ming-${kind} PRIVATE -O2 -fno-fast-math -ffp-contract=off)
    target_link_libraries(ming-${kind} PRIVATE onnxruntime dl)
endforeach()
add_library(ming-gpu-probe OBJECT "${MING_SOURCE}/native/gpu_probe.cpp")
target_include_directories(ming-gpu-probe PRIVATE "${ORT_INCLUDE}")
target_compile_definitions(ming-gpu-probe PRIVATE main=ming_gpu_probe_main)
target_sources(ming-synthesize PRIVATE $<TARGET_OBJECTS:ming-gpu-probe>)
target_compile_definitions(ming-synthesize PRIVATE VOICEBOX_GPU_PROBE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    foreach(kind IN ITEMS neon bf16)
        if(kind STREQUAL "neon")
            set(source portable_mmla_ops.cpp)
            set(arch armv8-a)
        else()
            set(source mmla_ops.cpp)
            set(arch armv8-a+bf16)
        endif()
        add_library(ming-mmla-${kind} SHARED "${MING_SOURCE}/native/${source}")
        target_include_directories(ming-mmla-${kind} PRIVATE "${ORT_INCLUDE}")
        target_compile_options(ming-mmla-${kind} PRIVATE -O3 -march=${arch} -fno-fast-math -ffp-contract=off)
    endforeach()
endif()
